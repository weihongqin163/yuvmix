```cpp
#include "video/mix_yuv_c.h"

#include <memory>
#include <new>
#include <stdexcept>
#include <utility>

#include "video/mix_yuv.h"

struct yuvmix_context {
    std::unique_ptr<yuvmix::MixYuvContext> impl;
};

namespace {

yuvmix_status ToCStatus(yuvmix::MixYuvStatus status) {
    switch (status) {
        case yuvmix::MixYuvStatus::kOk:
            return YUVMIX_STATUS_OK;
        case yuvmix::MixYuvStatus::kInvalidArgument:
            return YUVMIX_STATUS_INVALID_ARGUMENT;
        case yuvmix::MixYuvStatus::kBufferTooSmall:
            return YUVMIX_STATUS_BUFFER_TOO_SMALL;
        case yuvmix::MixYuvStatus::kOutOfMemory:
            return YUVMIX_STATUS_OUT_OF_MEMORY;
        case yuvmix::MixYuvStatus::kFontError:
            return YUVMIX_STATUS_FONT_ERROR;
        case yuvmix::MixYuvStatus::kLibyuvError:
            return YUVMIX_STATUS_LIBYUV_ERROR;
        case yuvmix::MixYuvStatus::kInternalError:
            return YUVMIX_STATUS_INTERNAL_ERROR;
    }
    return YUVMIX_STATUS_INTERNAL_ERROR;
}

}  // namespace

extern "C" yuvmix_status yuvmix_context_create(
    const yuvmix_config* config,
    yuvmix_context** context) {
    if (context == NULL) {
        return YUVMIX_STATUS_INVALID_ARGUMENT;
    }
    *context = NULL;
    if (config == NULL || config->font_path == NULL) {
        return YUVMIX_STATUS_INVALID_ARGUMENT;
    }

    try {
        yuvmix::MixYuvConfig cpp_config;
        cpp_config.font_path = config->font_path;
        cpp_config.font_face_index = config->font_face_index;
        cpp_config.font_size = config->font_size;
        cpp_config.osd_left = config->osd_left;
        cpp_config.osd_bottom = config->osd_bottom;

        std::unique_ptr<yuvmix::MixYuvContext> cpp_context;
        const yuvmix::MixYuvStatus status =
            yuvmix::MixYuvContext::Create(cpp_config, &cpp_context);
        if (status != yuvmix::MixYuvStatus::kOk) {
            return ToCStatus(status);
        }

        std::unique_ptr<yuvmix_context> wrapper(new yuvmix_context());
        wrapper->impl = std::move(cpp_context);
        *context = wrapper.release();
        return YUVMIX_STATUS_OK;
    } catch (const std::bad_alloc&) {
        return YUVMIX_STATUS_OUT_OF_MEMORY;
    } catch (const std::length_error&) {
        return YUVMIX_STATUS_OUT_OF_MEMORY;
    } catch (...) {
        return YUVMIX_STATUS_INTERNAL_ERROR;
    }
}

extern "C" void yuvmix_context_destroy(yuvmix_context* context) {
    delete context;
}
```

```c
static int init_i420(owned_i420* image,
                     uint32_t width,
                     uint32_t height,
                     uint8_t y,
                     uint8_t u,
                     uint8_t v) {
    size_t total_size;
    if (image == NULL || width == 0 || height == 0) {
        return 0;
    }
    memset(image, 0, sizeof(*image));
    image->width = width;
    image->height = height;
    image->y_size = (size_t)width * height;
    image->uv_size = (size_t)(width / 2) * (height / 2);
    total_size = image->y_size + 2 * image->uv_size;
    image->storage = (uint8_t*)malloc(total_size);
    if (image->storage == NULL) {
        return 0;
    }
    memset(image->storage, y, image->y_size);
    memset(image->storage + image->y_size, u, image->uv_size);
    memset(image->storage + image->y_size + image->uv_size,
           v, image->uv_size);
    return 1;
}

static void free_i420(owned_i420* image) {
    if (image != NULL) {
        free(image->storage);
        memset(image, 0, sizeof(*image));
    }
}

static yuvmix_i420_image const_view(const owned_i420* image) {
    yuvmix_i420_image view;
    memset(&view, 0, sizeof(view));
    view.y.data = image->storage;
    view.y.stride = (int)image->width;
    view.y.size = image->y_size;
    view.u.data = image->storage + image->y_size;
    view.u.stride = (int)(image->width / 2);
    view.u.size = image->uv_size;
    view.v.data = image->storage + image->y_size + image->uv_size;
    view.v.stride = (int)(image->width / 2);
    view.v.size = image->uv_size;
    view.width = image->width;
    view.height = image->height;
    return view;
}
```

```cpp
        case yuvmix::MixYuvStatus::kInternalError:
            return YUVMIX_STATUS_INTERNAL_ERROR;
    }
    return YUVMIX_STATUS_INTERNAL_ERROR;
}

bool ToCppFillMode(yuvmix_fill_mode mode, yuvmix::FillMode* cpp_mode) {
    if (cpp_mode == NULL) {
        return false;
    }
    switch (mode) {
        case YUVMIX_FILL_MODE_CONTAIN:
            *cpp_mode = yuvmix::FillMode::kContain;
            return true;
        case YUVMIX_FILL_MODE_COVER:
            *cpp_mode = yuvmix::FillMode::kCover;
            return true;
    }
    return false;
}

void CopyPlane(const yuvmix_const_plane& source,
               yuvmix::ConstPlane* destination) {
    destination->data = source.data;
    destination->stride = source.stride;
    destination->size = source.size;
}

void CopyPlane(const yuvmix_mutable_plane& source,
               yuvmix::MutablePlane* destination) {
    destination->data = source.data;
    destination->stride = source.stride;
    destination->size = source.size;
}

void CopyImage(const yuvmix_i420_image& source,
               yuvmix::I420ImageView* destination) {
    CopyPlane(source.y, &destination->y);
    CopyPlane(source.u, &destination->u);
    CopyPlane(source.v, &destination->v);
    destination->width = source.width;
    destination->height = source.height;
}

void CopyImage(const yuvmix_mutable_i420_image& source,
               yuvmix::MutableI420ImageView* destination) {
    CopyPlane(source.y, &destination->y);
    CopyPlane(source.u, &destination->u);
    CopyPlane(source.v, &destination->v);
    destination->width = source.width;
    destination->height = source.height;
}

extern "C" yuvmix_status yuvmix_context_create(
    const yuvmix_config* config,
    yuvmix_context** context) {
    if (context == NULL) {
        return YUVMIX_STATUS_INVALID_ARGUMENT;
    }
    *context = NULL;
    if (config == NULL || config->font_path == NULL) {
        return YUVMIX_STATUS_INVALID_ARGUMENT;
    }

    try {
        yuvmix::MixYuvConfig cpp_config;
        cpp_config.font_path = config->font_path;
        cpp_config.font_face_index = config->font_face_index;
        cpp_config.font_size = config->font_size;
        cpp_config.osd_left = config->osd_left;
        cpp_config.osd_bottom = config->osd_bottom;

        std::unique_ptr<yuvmix::MixYuvContext> cpp_context;
        const yuvmix::MixYuvStatus status =
            yuvmix::MixYuvContext::Create(cpp_config, &cpp_context);
        if (status != yuvmix::MixYuvStatus::kOk) {
            return ToCStatus(status);
        }

        std::unique_ptr<yuvmix_context> wrapper(new yuvmix_context());
        wrapper->impl = std::move(cpp_context);
        *context = wrapper.release();
        return YUVMIX_STATUS_OK;
    } catch (const std::bad_alloc&) {
        return YUVMIX_STATUS_OUT_OF_MEMORY;
    } catch (const std::length_error&) {
        return YUVMIX_STATUS_OUT_OF_MEMORY;
    } catch (...) {
        return YUVMIX_STATUS_INTERNAL_ERROR;
    }
}

extern "C" void yuvmix_context_destroy(yuvmix_context* context) {
    delete context;
}

extern "C" yuvmix_status yuvmix_mix(yuvmix_context* context,
                                    const yuvmix_source* sources,
                                    size_t source_count,
                                    yuvmix_output* output) {
    if (context == NULL || context->impl.get() == NULL || output == NULL ||
        (sources == NULL && source_count != 0)) {
        return YUVMIX_STATUS_INVALID_ARGUMENT;
    }

    try {
        std::vector<yuvmix::MixSource> cpp_sources(source_count);
        for (size_t i = 0; i < source_count; ++i) {
            yuvmix::MixSource& cpp_source = cpp_sources[i];
            CopyImage(sources[i].image, &cpp_source.image);
            cpp_source.destination.x = sources[i].destination.x;
            cpp_source.destination.y = sources[i].destination.y;
            cpp_source.destination.w = sources[i].destination.w;
            cpp_source.destination.h = sources[i].destination.h;
            cpp_source.display_name =
                sources[i].display_name == NULL ? "" : sources[i].display_name;
            if (!ToCppFillMode(sources[i].fill_mode,
                               &cpp_source.fill_mode)) {
                return YUVMIX_STATUS_INVALID_ARGUMENT;
            }
            cpp_source.is_highlight = sources[i].is_highlight != 0;
        }

        yuvmix::MixOutput cpp_output;
        CopyImage(output->image, &cpp_output.image);
        cpp_output.background_color.y = output->background_color.y;
        cpp_output.background_color.u = output->background_color.u;
        cpp_output.background_color.v = output->background_color.v;
        return ToCStatus(yuvmix::MixYuv(
            context->impl.get(),
            cpp_sources.empty() ? NULL : cpp_sources.data(),
            cpp_sources.size(), &cpp_output));
    } catch (const std::bad_alloc&) {
        return YUVMIX_STATUS_OUT_OF_MEMORY;
    } catch (const std::length_error&) {
        return YUVMIX_STATUS_OUT_OF_MEMORY;
    } catch (...) {
        return YUVMIX_STATUS_INTERNAL_ERROR;
    }
}
```
